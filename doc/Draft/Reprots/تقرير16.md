# 5. حالات سابقة يجب الاحتفاظ بها كذاكرة فشل

تم تثبيت الأخطاء السابقة بدل إخفائها:

## الخطأ 1 — مقارنة الملف الخطأ

المقارنة السابقة استخدمت `Current/PWA/main.html` بدل `Current/PWA/New-main`.

النتيجة: استنتاجات P154 لا يجوز إسقاطها تلقائيًا على الزوج الحالي.

## الخطأ 2 — false claim حول Idempotency-Key

سبق الادعاء أن Current لا يرسل `Idempotency-Key`.

المصدر الحالي يثبت أن تدفق الاستلام يحتوي بالفعل على:

```javascript

  'Content-Type':'application/json',
  'Authorization':'Bearer '+t,
  'Idempotency-Key':opId
}
```

ويرسل كذلك:

```javascript
operation_id:opId
```

إذًا الاستنتاج القديم كان غير صحيح وتم تصحيحه بالقراءة المباشرة.

## الخطأ 3 — الخلط بين DB fix وruntime proof

إصلاحات Production السابقة، ومنها Compatibility Projection الخاصة بـ`companies.main_branch_id/main_branch_code`، لا تعني تلقائيًا أن Cloudflare Worker الحي يقدم آخر artifact.

## الخطأ 4 — تجربة Receive Purchase السابقة

تم إجراء تجارب Transactional على منطق idempotency في الاستلام.

ظهر أن ترتيب فحص duplicate مقابل فحص المتبقي يحتاج عناية، وأن بناء operation identity فقط من state متغير مثل `qty_received_before` ليس هوية تشغيل مستقلة مثالية.

تم إجراء الاختبارات بطريقة لا تترك بيانات تشغيلية دائمة.

## الخطأ 5 — تعديل Production قبل اكتمال المقارنة الصحيحة

في المرحلة السابقة تم تنفيذ بعض إغلاقات Production المتعلقة بالـManual Voucher وغيرها قبل أن يتضح أن المهمة الحالية أصبحت **مقارنة Original vs New-main فقط**.

### التعديل

احذف **الدالة `RW_Audit_renderTab()` الحالية فقط**، ولا تحذف assignment في الأسفل.

أضف **فوق الدالة الحالية مباشرة** الكتلة التالية، ثم استبدل الدالة القديمة بالـwrapper الموجود في نهاية الكتلة:

```javascript
var RW_AUDIT_PAGE=1;
var RW_AUDIT_PAGE_SIZE=50;
var RW_AUDIT_TOTAL=0;
var RW_AUDIT_DATA=[];

async function RW_Audit_loadData(){
  var container=byId('audit-table-container');
  if(!container)return;

  safeHTML(container,'<div class="text-center py-10 text-gray-400">جاري التحميل...</div>');

  var from=(RW_AUDIT_PAGE-1)*RW_AUDIT_PAGE_SIZE;
  var to=from+RW_AUDIT_PAGE_SIZE-1;

  var query=supabase
    .from('audit_log')
    .select('*',{count:'exact'})
    .order('created_at',{ascending:false})
    .range(from,to);

  var searchEl=byId('audit-search');
  var actionEl=byId('audit-filter-action');
  var fromEl=byId('audit-filter-from');
  var toEl=byId('audit-filter-to');

  var searchVal=searchEl?String(searchEl.value||'').trim():'';
  var actionVal=actionEl?String(actionEl.value||'').trim():'';
  var fromDate=fromEl?String(fromEl.value||'').trim():'';
  var toDate=toEl?String(toEl.value||'').trim():'';

  if(searchVal){
    var safeSearch=searchVal.replace(/([%_.,()])/g,'\\$1');
    query=query.or('user_email.ilike.%'+safeSearch+'%,table_name.ilike.%'+safeSearch+'%,record_id.ilike.%'+safeSearch+'%');
  }
  if(actionVal)query=query.eq('action',actionVal);
  if(fromDate)query=query.gte('created_at',fromDate+'T00:00:00');
  if(toDate)query=query.lte('created_at',toDate+'T23:59:59.999');

  var res=await query;
  if(res.error){
    safeHTML(container,'<div class="text-center py-10 text-red-500">فشل تحميل سجل التدقيق</div>');
    console.error('RW_Audit_loadData',res.error);
    return;
  }

  RW_AUDIT_DATA=res.data||[];
  RW_AUDIT_TOTAL=res.count||0;
  RW_Audit_renderTable(RW_AUDIT_DATA);
}

function RW_Audit_renderTable(data){
  var container=byId('audit-table-container');
  if(!container)return;

  html+='<th class="p-2">المستخدم</th>';
  html+='<th class="p-2">الإجراء</th>';
  html+='<th class="p-2">الجدول</th>';
  html+='<th class="p-2">رقم السجل</th>';
  html+='<th class="p-2 text-center">تفاصيل</th>';
  html+='</tr></thead><tbody>';

  for(var i=0;i<data.length;i++){
    var log=data[i]||{};
    var dateStr=log.created_at?new Date(log.created_at).toLocaleString('ar-EG'):'';
    var actionLabel=log.action==='create'?'إنشاء':
      log.action==='update'?'تعديل':
      log.action==='delete'?'حذف':
      log.action==='login'?'دخول':

      String(log.action||'');

    html+='<tr class="border-t hover:bg-gray-50">';

function RW_Audit_renderPagination(){
  var pc=byId('audit-pagination-controls');
  if(!pc)return;

  var totalPages=Math.ceil(RW_AUDIT_TOTAL/RW_AUDIT_PAGE_SIZE);
  if(totalPages<=1){safeHTML(pc,'');return;}

  var html='<div class="flex flex-wrap items-center justify-center gap-2 text-sm">';
  html+='<span class="text-gray-500 ml-2">صفحة '+RW_AUDIT_PAGE+' من '+totalPages+' (إجمالي '+RW_AUDIT_TOTAL+' سجل)</span>';

  if(RW_AUDIT_PAGE>1){
    html+='<button type="button" onclick="RW_Audit_goPage('+(RW_AUDIT_PAGE-1)+')" class="px-3 py-1 border rounded-lg font-bold hover:bg-gray-100">السابق</button>';
  }

  for(var p=1;p<=totalPages;p++){
    if(p===RW_AUDIT_PAGE){
      html+='<span class="px-3 py-1 bg-blue-600 text-white rounded-lg font-bold">'+p+'</span>';
    }else if(p===1||p===totalPages||(p>=RW_AUDIT_PAGE-2&&p<=RW_AUDIT_PAGE+2)){
      html+='<button type="button" onclick="RW_Audit_goPage('+p+')" class="px-3 py-1 border rounded-lg font-bold hover:bg-gray-100">'+p+'</button>';
    }else if(p===RW_AUDIT_PAGE-3||p===RW_AUDIT_PAGE+3){
      html+='<span class="px-1">...</span>';
    }
  }

  if(RW_AUDIT_PAGE<totalPages){
    html+='<button type="button" onclick="RW_Audit_goPage('+(RW_AUDIT_PAGE+1)+')" class="px-3 py-1 border rounded-lg font-bold hover:bg-gray-100">التالي</button>';
  }

  html+='</div>';
  safeHTML(pc,html);
}
```
